<?php


 


use App\Models\Order;


 


// last week of paid orders, newest first


$orders = Order::query()


    ->where('status', 'paid')


    ->whereBetween('created_at', [now()->subDays(7), now()])


    ->with(['customer:id,name,email', 'items.product:id,name,sku'])


    ->withSum('items as units_sold', 'quantity')


    ->


    ->latest('created_at')->limit(5)->get();


 


$summary = $orders->groupBy('status')->map(fn (Order $order) => [


    'number'   => $order->number,


    'customer' => $order->customer->name,


    'total'    => $order->total_cents / 100,


    'paid_at'  => $order->paid_at?->diffForHumans(),


]);


 


return $summary->toArray();
